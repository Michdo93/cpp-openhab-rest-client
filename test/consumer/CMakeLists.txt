# Uses the installed library like an application does (CI: after cmake --install)
cmake_minimum_required(VERSION 3.16)
project(OpenHABRestClientCppConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenHABRestClientCpp 2 REQUIRED)
add_executable(consumer_static main.cpp)
target_link_libraries(consumer_static PRIVATE openhab::rest_client_cpp)
add_executable(consumer_shared main.cpp)
target_link_libraries(consumer_shared PRIVATE openhab::rest_client_cpp_shared)
